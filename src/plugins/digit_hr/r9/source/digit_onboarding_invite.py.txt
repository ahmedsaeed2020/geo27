import frappe
from frappe import _
from frappe.model.document import Document

GUARDED = ("token_hash", "invite_status", "expires_on", "submitted_on", "applied_by", "applied_on", "employee_onboarding",
	"personal_email", "cell_number", "current_address", "person_to_be_contacted", "emergency_phone_number", "relation")


class DIGITOnboardingInvite(Document):
	"""HIR-04: a single-use, expiring link a new joiner uses to enter contact / emergency details
	(digit_hr.api.onboarding_invite). Values stay on this record until HR applies them to the Employee.
	Invites are created and changed only by that API (it sets frappe.flags.digit_invite_action): a plain insert or save
	could otherwise plant a known token, bypass the site flag or alter what the joiner submitted."""

	def validate(self):
		if not frappe.flags.get("digit_invite_action"):
			before = self.get_doc_before_save()
			if self.is_new() or any((self.get(f) or "") != ((before.get(f) if before else None) or "") for f in GUARDED):
				frappe.throw(_("Use the invitation actions (create, apply, revoke)"), frappe.PermissionError)
		ob = frappe.db.get_value("Employee Onboarding", self.employee_onboarding, ["employee_name", "company"], as_dict=True) or {}
		self.employee_name, self.company = ob.get("employee_name"), ob.get("company")
