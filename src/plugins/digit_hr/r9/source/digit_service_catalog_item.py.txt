import frappe
from frappe import _
from frappe.model.document import Document


class DIGITServiceCatalogItem(Document):
	"""POR-03 service offered to employees: category, eligibility, SLA and the prompts the employee answers."""

	def validate(self):
		if self.sla_hours is not None and int(self.sla_hours or 0) < 0:
			frappe.throw(_("SLA hours cannot be negative"))
		seen = set()
		for row in self.eligibility:
			key = (row.based_on, row.value)
			if key in seen:
				frappe.throw(_("Row {0}: duplicate eligibility rule {1}: {2}").format(row.idx, _(row.based_on), row.value))
			seen.add(key)
