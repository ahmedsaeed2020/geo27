"""SHL-07 guided form tours: HR-owned, role-aware, bilingual walkthroughs of the native HR forms.

	GET /api/method/digit_hr.api.tours.for_form?doctype=<DocType>[&name=<record>]

Renderer = Frappe's own form tour (frappe.ui.form.FormTour on driver.js: Next / Previous / Done / Close). HR owns only the
definitions below and the filtering; desk forms start it from the form menu (public/js/digit_hr_tour.js), a host may render
the same steps. Rules:
- a tour is returned only to users holding one of its roles and able to read the DocType (and the record, when named);
- every step is checked against the LIVE meta (property setters included): a field that does not exist, is hidden, or sits
  on a permission level the user may not read is dropped — a restricted field is reported only as a count, never by name;
- steps whose visibility is a form expression carry it (depends_on) so the renderer shows them only when visible;
- tours never save, submit or change a document (no save_on_complete step), and carry no record data — text only.
Text is English source + Arabic from the app translations (ar.csv / ar.po)."""

DOCTYPES = frozenset(t[1] for t in TOURS)


def _both(text: str) -> dict:
	return {"en": text, "ar": _(text, lang="ar")}


def _visible_steps(meta, steps, readable_levels: list) -> tuple[list, dict]:
	out, skipped = [], {"absent": [], "hidden": [], "restricted": 0}
	for fieldname, title, description, position in steps:
		df = meta.get_field(fieldname)
		if not df:
			skipped["absent"].append(fieldname)
			continue
		if df.permlevel not in readable_levels:
			skipped["restricted"] += 1  # count only: a field the user may not read is not named
			continue
		if df.hidden:
			skipped["hidden"].append(fieldname)
			continue
		out.append({"fieldname": fieldname, "fieldtype": df.fieldtype, "label": _(df.label) if df.label else fieldname,
			"title": _both(title), "description": _both(description), "position": frappe.scrub(position).replace("_", "-"),
			"depends_on": df.depends_on or None, "is_table_field": 0})
	return out, skipped


@frappe.whitelist(methods=["GET"])
def for_form(doctype: str, name: str | None = None) -> dict:
	if not has_app_permission():
		frappe.throw(_("You do not have access to DIGIT HR"), frappe.PermissionError)
	if doctype not in DOCTYPES:
		return {"contract": "digit-hr-tours", "version": 1, "doctype": doctype, "state": "no_tour", "tours": []}
	if name:
		if not frappe.db.exists(doctype, name) or not frappe.has_permission(doctype, "read", doc=name):
			frappe.throw(_("You do not have permission to view this document"), frappe.PermissionError)
	elif not (frappe.has_permission(doctype, "read") or frappe.has_permission(doctype, "create")):
		frappe.throw(_("You do not have permission to view this document"), frappe.PermissionError)
	roles = set(frappe.get_roles())
	meta = frappe.get_meta(doctype)
	levels = meta.get_permlevel_access("read")
	if 0 not in levels:
		levels = [0, *levels]  # level 0 follows the DocType read check above
	tours = []
	for key, dt, section, tour_roles, title, steps in TOURS:
		if dt != doctype or not roles.intersection(tour_roles):
			continue
		visible, skipped = _visible_steps(meta, steps, levels)
		if not visible:
			continue
		tours.append({"key": key, "section": section, "title": _both(title), "roles": [r for r in tour_roles if r in roles],
			"steps": visible, "skipped": skipped, "save_on_complete": 0})
	return {"contract": "digit-hr-tours", "version": 1, "doctype": doctype, "name": name, "state": "ok" if tours else "no_tour",
		"can_write": bool(name and frappe.has_permission(doctype, "write", doc=name)) or (not name and frappe.has_permission(doctype, "create")),
		"renderer": "frappe.ui.form.FormTour", "buttons": ["Next", "Previous", "Done", "Close"], "tours": tours}
